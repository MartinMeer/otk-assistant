    // The backend endpoint that accepts file uploads
    const API_ENDPOINT = "/api/v1/upload-and-process";

    async function uploadAndProcess() {
        const fileInput = document.getElementById('dataFile');
        const typeSelect = document.getElementById('elementType');
        const submitBtn = document.getElementById('submitBtn');
        const statusEl = document.getElementById('status');
        const logOutput = document.getElementById('log-output');

        // 1. Validate that a file is selected
        if (fileInput.files.length === 0) {
            statusEl.textContent = 'Error: Please select a file first!';
            statusEl.style.color = 'red';
            return;
        }

        const file = fileInput.files[0];
        const elementType = typeSelect.value;

        // 2. Use FormData to package the file and metadata
        const formData = new FormData();
        formData.append('file', file); // 'file' is the name the backend will look for
        formData.append('elementType', elementType); // 'elementType' is the other parameter

        // 3. Update UI to show processing is in progress
        logOutput.textContent = 'Uploading file and starting process...';
        statusEl.textContent = `Processing ${file.name}... Please wait.`;
        statusEl.style.color = '#3498db';
        submitBtn.disabled = true;

        try {
            // 4. Send the data to the backend
            const response = await fetch(API_ENDPOINT, {
                method: 'POST',
                body: formData
                // IMPORTANT: Do NOT set the 'Content-Type' header yourself.
                // The browser will automatically set it to 'multipart/form-data'
                // with the correct boundary, which is required for file uploads.
            });

            // The backend should return the log as plain text
            const logText = await response.text();

            if (!response.ok) {
                // Handle server-side errors (e.g., 500 Internal Server Error)
                throw new Error(`Server returned status ${response.status}. Log:\n${logText}`);
            }

            // 5. Display the log from the server response
            logOutput.textContent = logText;
            statusEl.textContent = 'Processing complete!';
            statusEl.style.color = 'green';

        } catch (error) {
            console.error('Processing failed:', error);
            logOutput.textContent = `An error occurred:\n\n${error.message}`;
            statusEl.textContent = 'An error occurred. Check the log for details.';
            statusEl.style.color = 'red';
        } finally {
            // 6. Re-enable the button regardless of outcome
            submitBtn.disabled = false;
        }
    }
</script>

### How to Adapt Your Java Backend for This UI

This frontend sends data as `multipart/form-data`, which is the standard for file uploads. Your Java backend needs to be able to handle this.

1.  **Use a Web Framework:** You'll need a web server framework that can easily parse multipart requests. **Javalin**, **SparkJava**, or **Spring Boot** are all excellent choices.

2.  **Create a File Upload Endpoint:** Your endpoint (`POST /api/v1/upload-and-process`) will look different from a JSON endpoint.

    *   In **Javalin**, it's simple:
        ```java
        app.post("/api/v1/upload-and-process", ctx -> {
            UploadedFile uploadedFile = ctx.uploadedFile("file"); // "file" matches the FormData key
            String elementType = ctx.formParam("elementType"); // "elementType" matches the other key

            // You get an InputStream to read the file content
            try (InputStream fileContent = uploadedFile.content()) {
                // Here, you would adapt your DataProcessor to take an InputStream
                // instead of a file path. You would also need to capture logs.
                String logResult = myDataProcessor.process(fileContent, elementType);
                ctx.result(logResult); // Send the log back as a plain text response
            }
        });
        ```

    *   In **Spring Boot**, it's also very clean:
        ```java
        @PostMapping("/api/v1/upload-and-process")
        public ResponseEntity<String> handleFileUpload(
                @RequestParam("file") MultipartFile file,
                @RequestParam("elementType") String elementType) {

            // Spring gives you the file directly.
            // Again, adapt your processor and capture logs.
            String logResult = myDataProcessor.process(file.getInputStream(), elementType);
            return ResponseEntity.ok(logResult);
        }
        ```

3.  **Refactor for Logging:** Your `DataProcessor` and `SQLExecutor` currently log to the console with SLF4J. To make this work, you should modify them to append log messages to a `StringBuilder` and return it as a `String`. This string is what the web endpoint will send back to the browser to be displayed in the `<pre>` tag.
